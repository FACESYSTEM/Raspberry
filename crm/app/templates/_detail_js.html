{# インライン編集の共通処理。詳細画面から include する。 **押せると分かる手がかりを出す**（点線＋カーソル）。保存はその場で行い、 ボタンを押させない。ただし「押した瞬間に保存される」ことを画面に書く。 #} var CSRF = (document.querySelector('meta[name=csrf-token]') || {}).content || ''; function patch(field, value, done, fail) { fetch(PATCH_URL, { method: 'PATCH', headers: {'Content-Type': 'application/json', 'X-CSRFToken': CSRF}, body: JSON.stringify({field: field, value: value}) }).then(function (r) { return r.json().then(function (d) { return {ok: r.ok, d: d}; }); }).then(function (res) { if (!res.ok || !res.d.ok) { toast(res.d.message || '保存できませんでした'); if (fail) fail(); return; } toast('保存しました'); if (done) done(res.d.value); }).catch(function () { toast('通信できませんでした'); if (fail) fail(); }); } /* 文字の項目。押すと入力欄になり、離れると保存する */ document.querySelectorAll('.ed[data-field]').forEach(function (el) { el.addEventListener('click', function () { if (el.querySelector('input')) return; var was = el.dataset.value || ''; var i = document.createElement('input'); i.value = was; i.style.width = '100%'; el.textContent = ''; el.appendChild(i); i.focus(); /* 空のときの書き方は項目ごとに違う（「未記録」「不明」など）。 **空欄に「—」とだけ出さない**（規約 §12） */ var empty = el.dataset.empty || '—'; var finish = function () { var now = i.value.trim(); el.textContent = now || empty; if (now === was) return; patch(el.dataset.field, now, function (v) { el.dataset.value = v; el.textContent = v || empty; el.classList.add('changed'); }, function () { /* **保存できなかった値を、保存したように見せない** */ el.textContent = was || empty; }); }; i.addEventListener('blur', finish); i.addEventListener('keydown', function (e) { if (e.key === 'Enter') i.blur(); if (e.key === 'Escape') { i.value = was; i.blur(); } }); }); });
